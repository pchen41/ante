# Official checker for AlphaEvolve problem 59 (subsets of the grid with no isosceles triangles).
#
# `dist_sq` and `verify_construction` below are copied verbatim from the repository
# google-deepmind/alphaevolve_repository_of_problems, notebook
# experiments/subsets_of_the_grid_with_no_isosceles_triangles/subsets_of_the_grid_with_no_isosceles_triangles.ipynb
# (clone dated 2026-07-11). The imports and the small `evaluate` wrapper at the end are ours: the
# wrapper only adapts the function to the interface the sandbox calls. This file is third-party code
# and is only ever executed inside a network-blocked sandbox.
import itertools
from math import sqrt

def dist_sq(p1, p2):
    """Calculates the squared Euclidean distance between two points."""
    return (p1[0] - p2[0])**2 + (p1[1] - p2[1])**2

def verify_construction(points):
    """
    Verifies that for any three distinct points (a, b, c), the distance
    from a to b is not equal to the distance from b to c. This checks for
    both standard and degenerate/flat isosceles triangles.

    Args:
        points (list): A list of (x, y) tuples.

    Returns:
        bool: True if the construction is valid, False otherwise.
    """
    print("Starting verification... (this may take a moment)")
    # Generate all ordered permutations of 3 points from the list
    for a, b, c in itertools.permutations(points, 3):
        # Check if the distance from a to b is the same as b to c.
        if dist_sq(a, b) == dist_sq(b, c):
            d = sqrt(dist_sq(a,b))
            print(f"Verification FAILED: Found a forbidden isosceles configuration.")
            print(f"Points: a={a}, b={b}, c={c}")
            print(f"dist(a,b) = dist(b,c) = {d:.2f}")
            return False

    # If the loop completes, no such configuration was found
    return True


# --- wrapper (ours) -------------------------------------------------------------------------
def evaluate(data):
    points = [tuple(int(v) for v in p) for p in data["points"]]
    if not verify_construction(points):
        raise ValueError("verify_construction returned False")
    return float(len(points))
