   def build(n, R, m, gens, light_k=None):            # gens: list of ('t',i,j) transpositions / ('x',mask) translations
       N = 1 << n; half = N >> 1; FULL = N - 1
       pair = lambda v: v if v < half else v ^ FULL
       balls = [b for b in range(N) if bin(b).count('1') <= R]
       cnf = CNF()
       for x in range(half):                          # covering clauses, points x < 2^(n-1)
           cnf.append(sorted({pair(x ^ b) + 1 for b in balls}))
       top = half
       card = CardEnc.atmost(lits=list(range(1, half + 1)), bound=m, top_id=top, encoding=EncType.seqcounter)
       cnf.extend(card.clauses); top = max(top, card.nv)
       cnf.append([1])                                # pair {0, all-ones} present
       pw = [min(bin(p).count('1'), n - bin(p).count('1')) for p in range(half)]
       if light_k is not None:
           eq = CardEnc.equals(lits=[p + 1 for p in range(1, half) if pw[p] <= 3], bound=light_k, top_id=top, encoding=EncType.seqcounter)
           cnf.extend(eq.clauses); top = max(top, eq.nv)
       for g in gens:                                 # lex-MAX leader: y >=_lex g(y), chain with flags
           sig = (lambda p: pair(p ^ g[1])) if g[0] == 'x' else (lambda p, i=g[1], j=g[2]: pair((p & ~((1 << i) | (1 << j))) | (((p >> i) & 1) << j) | (((p >> j) & 1) << i)))
           e_prev = None
           for k in range(half):
               sk = sig(k)
               if sk == k: continue
               yk, ysk = k + 1, sk + 1
               if e_prev is None: cnf.append([yk, -ysk]); top += 1; e = top; cnf.append([-yk, -ysk, e]); cnf.append([yk, ysk, e])

       return cnf
